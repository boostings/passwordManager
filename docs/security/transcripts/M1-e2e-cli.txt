#   expect run.exp "$PWD" <tmp>/v/vault.pmv transcript.txt

# run.exp sets PM_SKIP_BUILD=1, spawns `scripts/pm --vault <tmp>/v/vault.pmv <command>` on a
# pseudo-terminal for each step, answers each prompt with `send`, and logs the terminal output

# it as drwx------ and wrote vault.pmv, vault.pmv.bak.1 and vault.pmv.lock as -rw-------.
#
# Passphrase prompts are read with Console.readPassword (echo off), so the typed passphrase,
# login password and wrong passphrase do not appear below: `grep -c` for all three over the raw
# log returned 0. Title, username, URLs and tags are echoed because they are not secret.
# The recovery key of this throwaway vault (deleted after the run) is redacted to keep secret
# scanners quiet; the CLI printed a 56-character key as 8 dash-separated groups of 7.
#
# run.exp (passphrases here are throwaway test values):
#   set timeout 120
#   set root [lindex $argv 0]; set vault [lindex $argv 1]
#   log_file -noappend [lindex $argv 2]

#   proc done {} { global spawn_id; expect eof
